[docs]    def apply(
        self,
        current_user: UserPublic,
        session: Session,
        commit: LabelsetCommit,
    ) -> uuid.UUID:
        params = self.params
        base_op_type = params.builders.create_op
        insert_hash_to_distance_perf = params.insert_hash_to_distance_perf

        base_domain = base_op_type.get_domain()
        assert isinstance(base_domain, LabelEntityDomain)

        with PerformanceCounter(
            init_seconds=insert_hash_to_distance_perf.total_seconds
        ) as create_perf:
            base_params = params.builders.create()
            add_perf = PerformanceCounter()

            entity = base_domain.create_data(
                current_user=current_user,
                session=session,
                data=base_op_type.get_data(commit, base_params),
                add_perf=add_perf,
            )

        self.params.history.add_create_history(
            step=self.params.step,
            num_entities=self.params.num_entities,
            num_elements=self.params.num_elements,
            create_target="entity",
            insert_hash_to_distance_perf=insert_hash_to_distance_perf,
            add_perf=add_perf,
            create_perf=create_perf,
        )

        return entity.id

                    element_domain = element_create_op.get_domain()
                    element_op_params = [
                        op_builders.element.create(entity_id)
                        for entity_id in random.choices(

                        )
                    ]

                    element_domain.bulk_create_datas(
                        current_user=current_user,

                        data=[element_create_op.get_data(commit, p) for p in element_op_params],
                    )

    group_id: int,

[docs]def split_rng_seeds(base_seed: int | None, stream_count: int) -> list[int | None]:
    """
    Splits one base seed into independent seeds for separate random streams.

    A run uses one stream for the setup phase that precedes the concurrent users and one
    stream per user, so that no user replays another user's draws while the setup phase
    still stays identical across runs that share a base seed.

    Parameters
    ----------
    base_seed : int or None
        The seed the caller was given. `None` leaves every stream unseeded.
    stream_count : int
        The number of streams to derive seeds for.

    Returns
    -------
    list of (int or None)
        One seed per stream; all `None` when `base_seed` is `None`.
    """
    if base_seed is None:
        return [None] * stream_count

    master = random.Random(base_seed)

    # 32 bits keeps every derived seed valid for NumPy's legacy generator as well as the
    # standard library's.
    return [master.getrandbits(32) for _ in range(stream_count)]




[docs]def seed_benchmark_rng(stream_seed: int | None) -> None:
    """
    Applies one stream's seed to every generator the workload draws from.

    The workload pulls both from the standard-library module stream and from NumPy's legacy
    global stream, so a run is reproducible only when both are seeded. Each process must
    call this itself: a forked worker inherits the parent's generator state, which would
    otherwise leave every concurrent user drawing the identical sequence.

    Parameters
    ----------
    stream_seed : int or None
        The stream's seed, as returned by `split_rng_seeds`. `None` leaves the process's
        current seeding untouched.
    """
    if stream_seed is None:
        return

    random.seed(stream_seed)
    np.random.seed(stream_seed)  # noqa: NPY002
